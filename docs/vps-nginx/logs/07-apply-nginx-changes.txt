# SNI-мультиплексор на публичном 443.
# Пояснение:
#   - TLS + SNI tyoung.idivles.ru  → локальный HTTPS-сайт портала (Young/Sochi)
#   - TLS + SNI v1.idivles.ru      → локальный HTTPS vhost (заглушка/маска + gRPC /gun)
#   - нет TLS / пустой SNI / другой SNI → Xray VLESS+gRPC (security=none клиенты)
#
# Это позволяет одновременно держать сайт и VPN на одном порту 443.
map $ssl_preread_server_name $yp_backend {
    tyoung.idivles.ru 127.0.0.1:8443;
    v1.idivles.ru     127.0.0.1:8445;
    default           127.0.0.1:10443;
}
server {
    listen 443 reuseport;
    listen [::]:443 reuseport;
    proxy_pass $yp_backend;
    ssl_preread on;
    proxy_connect_timeout 5s;
    proxy_timeout 86400s;
}
